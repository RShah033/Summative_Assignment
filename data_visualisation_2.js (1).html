/**
 *  Margin
 * @var {number}
 * 
 * Width
 * @var {number} width
 * Height
 * @var {number} height
 * 
 */

var margin = {top: 10, right: 30, bottom: 30, left: 60},
    width = 460 - margin.left - margin.right,
    height = 400 - margin.top - margin.bottom;

/**append the svg object to the body of the page */
var svg = d3.select("#temp_graph")
  .append("svg")
    .attr("width", width + margin.left + margin.right)
    .attr("height", height + margin.top + margin.bottom)
  .append("g")
    .attr("transform",
          "translate(" + margin.left + "," + margin.top + ")");

/**Read the data */

/**
 * A method that is on d3 which allows the data from the CSV file to be converted into a graph using d3.js
 * @function d3.csv */

/** 
 * Function to collect the data from the csv file (Precipitation) so that actions can be performed on it
 * @function data
 */
d3.csv("./Temperatures.csv", function(data) {

  /**
             * variable to define the values that go onto the x-axis
             * @var x
            */
  var x = d3.scaleLinear()
    .domain([1989, 2002])
    .range([ 0, width ]);
  svg.append("g")
    .attr("transform", "translate(0," + height + ")")
    .call(d3.axisBottom(x));

/**Add text label for x-axis 
 * @var x
*/
svg.append("text")
    .attr("class", "x label")
    .attr("text-anchor", "end")
    .attr("x", width)
    .attr("y", height - 25)
    .text("Year");

   
/**
     * variable to define the values that go onto the y-axis
     * @var y 
     * */
  var y = d3.scaleLinear()
    .domain([8.5, 11])
    .range([ height, 0]);
  svg.append("g")
    .call(d3.axisLeft(y));

/**
 * Add the text for the y-axis label
 * @var y
 */
    svg.append("text")
    .attr("class", "y label")
    .attr("text-anchor", "end")
    .attr("y", -45)
    .attr("dy", ".75em")
    .attr("transform", "rotate(-90)")
    .text("Average Annual Temperature (degrees Celcius)");

    svg.append('g')
    .selectAll("dot")
    .data(data)
    .enter()
    .append("circle")
      .attr("cx", function (d) { return x(d.Year2); } )
      .attr("cy", function (d) { return y(d.Annual2); } )
      .attr("r", 5)
      .style("fill", "#ff160c");


    /**
     * Add brushing
     * @param #dataviz_brushing */
    d3.select("#dataviz_brushing")
        .call( d3.brush()                     // Add the brush feature using the d3.brush function
        .extent( [ [0,0], [400,400] ] )       // initialise the brush area: start at 0,0 and finishes at width,height: it means I select the whole graph area
);

});
